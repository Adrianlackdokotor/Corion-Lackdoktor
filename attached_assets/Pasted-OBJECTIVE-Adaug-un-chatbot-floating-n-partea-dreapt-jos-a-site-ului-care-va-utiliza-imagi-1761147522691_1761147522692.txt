### ⚙️ 2️⃣ **Integrarea cu API-ul GPT**

1. Creează un fișier `CorionAgent.ts` în `agents/` pentru a trimite cereri la backend:
   ```ts
   export async function askCorionAgent(prompt: string) {
     const res = await fetch("/api/ai", {
       method: "POST",
       headers: { "Content-Type": "application/json" },
       body: JSON.stringify({ prompt }),
     });
     const data = await res.json();
     return data.reply;
   }
   ```

2. Creează un backend API în `server/routes/ai.ts` care gestionează cererea către OpenAI GPT:
   ```ts
   import express from "express";
   import fetch from "node-fetch";
   import dotenv from "dotenv";

   dotenv.config();
   const router = express.Router();
   const OPENAI_API_KEY = process.env.OPENAI_API_KEY_CORIONLACKDOKTOR;

   router.post("/", async (req, res) => {
     const { prompt } = req.body;

     const response = await fetch("https://api.openai.com/v1/chat/completions", {
       method: "POST",
       headers: {
         "Content-Type": "application/json",
         "Authorization": `Bearer ${OPENAI_API_KEY}`,
       },
       body: JSON.stringify({
         model: "gpt-4o-mini",
         messages: [
           { role: "system", content: "You are Corion's assistant." },
           { role: "user", content: prompt },
         ],
       }),
     });

     const data = await response.json();
     res.json({ reply: data.choices?.[0]?.message?.content ?? "Sorry, I couldn't get that." });
   });

   export default router;
   ```

### 🧩 3️⃣ **Actualizarea frontend pentru interacțiuni cu GPT**

1. În `ChatWidget.tsx`, actualizează pentru a trimite mesaje către GPT și a le afișa în chat:
   ```tsx
   import { useState } from "react";
   import { askCorionAgent } from "@/agents/CorionAgent";
   import ChatIcon from "../public/chat-icon.png";

   export default function ChatWidget() {
     const [isOpen, setIsOpen] = useState(false);
     const [messages, setMessages] = useState<{ sender: string, text: string }[]>([]);
     const [input, setInput] = useState("");

     const toggleChat = () => setIsOpen(!isOpen);

     const handleSend = async () => {
       const userMessage = { sender: "user", text: input };
       setMessages([...messages, userMessage]);

       const response = await askCorionAgent(input);
       setMessages([...messages, userMessage, { sender: "ai", text: response }]);

       setInput("");
     };

     return (
       <div>
         <div
           onClick={toggleChat}
           className="fixed bottom-5 right-5 z-50 cursor-pointer"
         >
           <img src={ChatIcon} alt="Chat with us" className="w-12 h-12 rounded-full" />
         </div>

         {isOpen && (
           <div className="fixed bottom-20 right-5 w-80 bg-darkBackground text-white p-4 rounded-lg shadow-lg">
             <div className="flex items-center justify-between">
               <span>Chat with Corion AI</span>
               <button onClick={toggleChat} className="text-xl">❌</button>
             </div>
             <div className="mt-2">
               {messages.map((msg, idx) => (
                 <div key={idx} className={msg.sender === "ai" ? "text-primary" : "text-gray-300"}>
                   {msg.text}
                 </div>
               ))}
               <input
                 value={input}
                 onChange={(e) => setInput(e.target.value)}
                 className="w-full p-2 rounded-xl bg-gray-800 text-white"
                 placeholder="Type your message..."
               />
               <button onClick={handleSend} className="bg-primary text-white p-2 rounded-xl mt-2 w-full">
                 Send
               </button>
             </div>
           </div>
         )}
       </div>
     );
   }
   ```
